cmake_minimum_required(VERSION 3.20)
project(HyperGUI VERSION 1.0.0 LANGUAGES CXX)
option(HYPER_BUILD_DEMO "Build the Windows DirectX11 demo" ON)
set(HYPER_IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/imgui" CACHE PATH "Dear ImGui source directory")
add_library(hyper_gui STATIC src/hyper_gui.cpp)
target_compile_features(hyper_gui PUBLIC cxx_std_17)
target_include_directories(hyper_gui PUBLIC src "${HYPER_IMGUI_DIR}")
if(MSVC)
 target_compile_options(hyper_gui PRIVATE /utf-8 /W4)
endif()
if(HYPER_BUILD_DEMO)
 if(NOT WIN32)
  message(FATAL_ERROR "The demo requires Windows. Disable HYPER_BUILD_DEMO for the portable component.")
 endif()
 add_executable(hyper_demo demo/main.cpp "${HYPER_IMGUI_DIR}/imgui.cpp" "${HYPER_IMGUI_DIR}/imgui_draw.cpp" "${HYPER_IMGUI_DIR}/imgui_tables.cpp" "${HYPER_IMGUI_DIR}/imgui_widgets.cpp" "${HYPER_IMGUI_DIR}/backends/imgui_impl_win32.cpp" "${HYPER_IMGUI_DIR}/backends/imgui_impl_dx11.cpp")
 target_include_directories(hyper_demo PRIVATE "${HYPER_IMGUI_DIR}/backends")
 target_link_libraries(hyper_demo PRIVATE hyper_gui d3d11 dxgi d3dcompiler dwmapi windowscodecs ole32 advapi32)
 if(MSVC)
  target_compile_options(hyper_demo PRIVATE /utf-8)
 endif()
endif()

option(HYPER_BUILD_TESTS "Build local profile selection tests" OFF)
if(HYPER_BUILD_TESTS)
 enable_testing()
 add_executable(steam_profile_test tests/steam_profile_test.cpp)
 target_include_directories(steam_profile_test PRIVATE demo)
 target_compile_features(steam_profile_test PRIVATE cxx_std_17)
 add_test(NAME steam_profile_selection COMMAND steam_profile_test)
endif()
